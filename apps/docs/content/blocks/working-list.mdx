## Basic usage

Customize the columns, filters, actions, and records table to fit your users’ workflow.

```tsx
import {
    WorkingList,
    WorkingListColumnSelector,
    WorkingListFilterControls,
    WorkingListTable,
    WorkingListToolbar,
    WorkingListToolbarActions,
    WorkingListViewSwitcher,
} from '@/components/working-list'

export const BasicWorkingList = () => (
    <WorkingList
        programId="IpHINAT79UW"
        orgUnitId="y77LiPqLMoq"
        defaultPagination={{ page: 1, pageSize: 10 }}
    >
        <WorkingListViewSwitcher />
        <WorkingListToolbar>
            <WorkingListColumnSelector />
            <WorkingListFilterControls />
            <WorkingListToolbarActions />
        </WorkingListToolbar>
        <WorkingListTable />
    </WorkingList>
)
```
